cmake_minimum_required(VERSION 3.24)
project(ui_debug_tests LANGUAGES CXX)
find_package(Qt6 6.2 REQUIRED COMPONENTS Core)
add_executable(ui_debug_tests debug_tests.cpp ../../src/gui/ui/screens/developer_ui/DebugController.cpp)
target_include_directories(ui_debug_tests PRIVATE ../../src)
target_compile_features(ui_debug_tests PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(ui_debug_tests PRIVATE /W4 /WX)
endif()
enable_testing()
add_test(NAME ui_debug_tests COMMAND ui_debug_tests)
add_test(NAME ui_debug_contract COMMAND ${CMAKE_COMMAND} -DROOT=${CMAKE_CURRENT_LIST_DIR}/../.. -P ${CMAKE_CURRENT_LIST_DIR}/verify-debug-contract.cmake)
add_executable(ui_debug_release_registry release_registry_tests.cpp ../../src/gui/ui/state/UiRegistries.cpp)
target_include_directories(ui_debug_release_registry PRIVATE ../../src)
target_compile_features(ui_debug_release_registry PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(ui_debug_release_registry PRIVATE /W4 /WX)
endif()
add_test(NAME ui_debug_release_registry COMMAND ui_debug_release_registry)
add_library(ui_debug_qt_adapter OBJECT ../../src/gui/ui/controllers/developer_ui/DebugQtDataAdapter.cpp)
target_include_directories(ui_debug_qt_adapter PRIVATE ../../src)
target_compile_features(ui_debug_qt_adapter PRIVATE cxx_std_20)
target_link_libraries(ui_debug_qt_adapter PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(ui_debug_qt_adapter PRIVATE /W4 /WX)
endif()
add_library(ui_debug_qt_port OBJECT ../../src/gui/ui/controllers/developer_ui/DebugQtCommandPort.cpp)
