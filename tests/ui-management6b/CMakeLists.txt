cmake_minimum_required(VERSION 3.16)
project(IngnomiaUiManagement6BTests LANGUAGES CXX)
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(ui_management6b_controller controller_tests.cpp "${ROOT}/src/gui/ui/screens/management6b/Management6BController.cpp")
target_compile_features(ui_management6b_controller PRIVATE cxx_std_20)
target_include_directories(ui_management6b_controller PRIVATE "${ROOT}/src")
if(MSVC)
  target_compile_options(ui_management6b_controller PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(ui_management6b_controller PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ui_management6b_controller COMMAND ui_management6b_controller)
add_test(NAME ui_management6b_rml_contract COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_LIST_DIR}/verify-management6b-rml.cmake")
add_test(NAME ui_management6b_integration_contract COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_LIST_DIR}/verify-management6b-integration.cmake")
