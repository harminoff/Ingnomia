cmake_minimum_required(VERSION 3.16)
project(IngnomiaUiHudTests LANGUAGES CXX)
include(CTest)
find_package(Qt6 REQUIRED COMPONENTS Core)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(ui_hud_controller controller_tests.cpp
  "${ROOT}/src/gui/ui/screens/hud/HudController.cpp"
  "${ROOT}/src/gui/ui/controllers/hud/LegacyToolActionAdapter.cpp")
target_compile_features(ui_hud_controller PRIVATE cxx_std_20)
target_include_directories(ui_hud_controller PRIVATE "${ROOT}/src")
target_link_libraries(ui_hud_controller PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(ui_hud_controller PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(ui_hud_controller PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
set(QT_TEST_ENV "PATH=$<TARGET_FILE_DIR:Qt6::Core>;$ENV{PATH}")
add_test(NAME ui_hud_controller COMMAND "${CMAKE_COMMAND}" -E env "${QT_TEST_ENV}" "$<TARGET_FILE:ui_hud_controller>")
add_executable(ui_hud_fifo fifo_tests.cpp
  "${ROOT}/src/gui/ui/screens/hud/HudController.cpp")
target_compile_features(ui_hud_fifo PRIVATE cxx_std_20)
target_include_directories(ui_hud_fifo PRIVATE "${ROOT}/src")
target_link_libraries(ui_hud_fifo PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(ui_hud_fifo PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(ui_hud_fifo PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ui_hud_fifo COMMAND "${CMAKE_COMMAND}" -E env "${QT_TEST_ENV}" "$<TARGET_FILE:ui_hud_fifo>")
add_executable(ui_hud_build_actions build_action_tests.cpp
  "${ROOT}/src/gui/ui/screens/hud/HudController.cpp")
target_compile_features(ui_hud_build_actions PRIVATE cxx_std_20)
target_include_directories(ui_hud_build_actions PRIVATE "${ROOT}/src")
target_link_libraries(ui_hud_build_actions PRIVATE Qt6::Core)
if(MSVC)
  target_compile_options(ui_hud_build_actions PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(ui_hud_build_actions PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ui_hud_build_actions COMMAND "${CMAKE_COMMAND}" -E env "${QT_TEST_ENV}" "$<TARGET_FILE:ui_hud_build_actions>")

  "${ROOT}/src/gui/ui/screens/hud/HudController.cpp")

endif()
add_test(NAME ui_hud_unavailable COMMAND "${CMAKE_COMMAND}" -E env "${QT_TEST_ENV}" "$<TARGET_FILE:ui_hud_unavailable>")
