cmake_minimum_required(VERSION 3.16)
project(IngnomiaUiInspectorTests LANGUAGES CXX)
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(ui_inspector_controller controller_tests.cpp "${ROOT}/src/gui/ui/screens/inspector/InspectorController.cpp")
target_compile_features(ui_inspector_controller PRIVATE cxx_std_20)
target_include_directories(ui_inspector_controller PRIVATE "${ROOT}/src")
if(MSVC)
  target_compile_options(ui_inspector_controller PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(ui_inspector_controller PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ui_inspector_controller COMMAND ui_inspector_controller)
add_executable(ui_inspector_skills skills_tests.cpp "${ROOT}/src/gui/ui/screens/inspector/InspectorController.cpp")
target_compile_features(ui_inspector_skills PRIVATE cxx_std_20)
target_include_directories(ui_inspector_skills PRIVATE "${ROOT}/src")
if(MSVC)
  target_compile_options(ui_inspector_skills PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(ui_inspector_skills PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ui_inspector_skills COMMAND ui_inspector_skills)
add_test(NAME ui_inspector_rml_contract COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_LIST_DIR}/verify-inspector-rml.cmake")
add_test(NAME ui_inspector_integration_contract COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_LIST_DIR}/verify-inspector-integration.cmake")

find_package(Qt6 6.2 REQUIRED COMPONENTS Core Xml)
add_executable(water_flow_tests water_flow_tests.cpp "${ROOT}/src/game/waterflowsolver.cpp")
target_compile_features(water_flow_tests PRIVATE cxx_std_20)
target_include_directories(water_flow_tests PRIVATE "${ROOT}/src")
target_link_libraries(water_flow_tests PRIVATE Qt6::Core Qt6::Xml)
if(MSVC)
  target_compile_options(water_flow_tests PRIVATE /W4 /WX /permissive-)
else()
  target_compile_options(water_flow_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME water_flow_tests COMMAND water_flow_tests)
