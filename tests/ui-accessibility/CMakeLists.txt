cmake_minimum_required(VERSION 3.16)
project(IngnomiaUiAccessibilityTests LANGUAGES CXX)
include(CTest)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(ui_accessibility_tests accessibility_tests.cpp "${ROOT}/src/gui/ui/localization/UiText.cpp" "${ROOT}/src/gui/ui/accessibility/UiAccessibility.cpp")
target_compile_features(ui_accessibility_tests PRIVATE cxx_std_20)
target_include_directories(ui_accessibility_tests PRIVATE "${ROOT}/src")
if(MSVC)
 target_compile_options(ui_accessibility_tests PRIVATE /W4 /WX /permissive- /utf-8)
else()
 target_compile_options(ui_accessibility_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ui_accessibility_tests COMMAND ui_accessibility_tests)
add_test(NAME ui_accessibility_contract COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_LIST_DIR}/verify-accessibility.cmake")
add_executable(ui_workbench_tests workbench_tests.cpp "${ROOT}/src/gui/ui/navigation/WorkbenchCoordinator.cpp" "${ROOT}/src/gui/ui/state/UiNavigation.cpp" "${ROOT}/src/gui/ui/state/UiRegistries.cpp" "${ROOT}/src/gui/ui/accessibility/UiAccessibility.cpp")
target_compile_features(ui_workbench_tests PRIVATE cxx_std_20)
target_include_directories(ui_workbench_tests PRIVATE "${ROOT}/src")
if(MSVC)
 target_compile_options(ui_workbench_tests PRIVATE /W4 /WX /permissive- /utf-8)
else()
 target_compile_options(ui_workbench_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME ui_workbench_tests COMMAND ui_workbench_tests)
